\begin{tikzpicture}[
    >=Stealth,
    font=\sffamily\small,
    % Flat fill + heavy tinted border carries node identity; the blur shadow is barely there
    % (opacity 11) so a card lifts off the page without reading as a 2005 gradient. Rounded
    % rectangles also edge-align, which is what lets the three bands line up -- ellipses cannot.
    card/.style={draw, line width=1.5pt, rounded corners=3pt, align=center, fill=white,
                 minimum height=8.5mm, inner xsep=2.4mm, font=\sffamily\small\bfseries,
                 blur shadow={shadow blur steps=8, shadow xshift=0.5pt, shadow yshift=-0.8pt,
                              shadow opacity=11, shadow blur radius=1.3pt}},
    pic/.style={card, inner sep=1pt, minimum height=0pt},       % image cards: no text padding
    lat/.style ={card, draw=diagLatent!70!black, text=diagLatent!25!black},
    prb/.style ={card, draw=diagProbe!75!black,  text=diagProbe!30!black},
    law/.style ={card, draw=diagLaw!70!black,    text=diagLaw!25!black},
    pln/.style ={card, draw=diagPlan!70!black,   text=diagPlan!25!black},
    % Ordinary dataflow is deliberately recessive: thin, grey. The gaps must win the page.
    flow/.style={-{Stealth[length=5.5pt]}, line width=0.9pt, draw=inkmute!60},
    % THE MOTIF: the two gaps are the conceptual punchline, so they are the heaviest marks here,
    % and they are the only two horizontal arrows in the middle of the figure -- mirrored.
    gap/.style={line width=2.4pt, dash pattern=on 5.5pt off 3pt, -{Stealth[length=7pt]}},
    % Solid-colour capsule, white text, sitting just ABOVE the line it names so the dashes show.
    caps/.style={fill=#1, text=white, rounded corners=3.5pt, inner xsep=4pt, inner ysep=1.9pt,
                 font=\sffamily\scriptsize\bfseries},
    band/.style={font=\sffamily\scriptsize\bfseries, text=inkmute!80, anchor=west},
    plab/.style={font=\sffamily\scriptsize\bfseries, align=center, anchor=north},
]

%======================= the two panels: world (left) and planner =======================%

      {\includegraphics[width=25mm]{inset_world_p.png}};
\node[plab, text=diagEnv!30!black, below=0.9mm of world] {Isaac Lab pusher};

%======================= LAW band: source text and the CLASSICAL isomorphism =======================%

      (statute) -- node[caps=inkmute!70, above=0.4mm] {isomorphism} (rules);

%======================= PERCEPTION band: world -> atoms (the GROUNDING gap) =======================%
\node[lat] (wm)     at (6.2,0.3)  {World model $\psi$};
\node[prb] (probes) at (9.9,0.3)  {Probes $\phi$};

\draw[flow] (world) -- (wm);
\draw[flow] (wm) -- (probes);
\draw[gap, draw=gapblue] (probes) -- node[caps=gapblue, above=0.4mm] {Grounding gap} (atoms);

%======================= CONTROL band: verdict -> constraint (the ONTOLOGICAL gap) =======================%
\node[law] (verdict) at (13.6,-2.2) {Verdict};

\draw[flow] (cons) -- (planner);

%======================= the closed loop, made explicit =======================%

      node[midway, above=0.5pt, font=\sffamily\scriptsize, text=inkmute] {action} |- (world.south west);

%======================= band labels =======================%
\node[band] at (5.55,3.62)  {\textsc{law}};
